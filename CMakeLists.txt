cmake_minimum_required(VERSION 3.20)

if(APPLE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum macOS version")
endif()

project(sec_epp VERSION 0.1.0 LANGUAGES CXX)

if(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 20)
    set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
    set(CMAKE_OBJCXX_EXTENSIONS OFF)
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(SEC_BUILD_TESTS "Build unit tests" ON)
option(SEC_BUILD_SERVICE "Build the Windows service host" ${WIN32})

function(sec_apply_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()

add_subdirectory(src/common)
add_subdirectory(src/scanner)
add_subdirectory(src/agent)
add_subdirectory(src/behavior)
if(NOT WIN32)
    add_subdirectory(src/sensor)
endif()
add_subdirectory(src/cli)
add_subdirectory(src/agentd)
if(NOT WIN32)
    add_subdirectory(src/sensord)
endif()

if(APPLE)
    add_subdirectory(src/monitor)
    add_subdirectory(src/macapp)
endif()

if(SEC_BUILD_SERVICE)
    if(NOT WIN32)
        message(FATAL_ERROR "SEC_BUILD_SERVICE requires Windows")
    endif()
    add_subdirectory(src/service)
endif()

if(SEC_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
